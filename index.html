Welcome — let's get you set up 👋
Outreach Command is a private, offline CRM for freelancers doing cold outreach. Everything stays on your device — no accounts, no subscriptions.
1
Add prospects — anyone you plan to reach out to. Assign them a Status as you progress.
2
Status pipeline: New → Contacted → Replied → Interested → Proposal Sent → Converted → Not Interested
3
Hook = the type of opening message you used (e.g. Warm check-in, Value-first, Direct). Used to track which message style gets more replies.
4
Log activities on each prospect — messages sent, replies received, follow-ups, calls, notes. The dashboard tracks everything automatically.

Gmail Sync
Auto-read your Gmail threads and update prospect statuses. Free via Google Apps Script.
Connect Your Gmail — Free Setup (~3 min)
Uses Google Apps Script: runs in your own Google account, reads your Gmail directly. No third party ever sees your emails.
1
Go to script.google.com → New Project → paste the script below → Deploy → New Deployment → Web App → "Execute as: Me" → "Who has access: Only myself" → Deploy → copy the URL.
function doGet(e) {
  var emails = JSON.parse(e.parameter.emails || '[]');
  var days   = parseInt(e.parameter.days   || '30');
  var cutoff = new Date();
  cutoff.setDate(cutoff.getDate() - days);
  var after  = Utilities.formatDate(cutoff, 'GMT', 'yyyy/MM/dd');
  var out    = {};
  emails.forEach(function(email) {
    if (!email) return;
    var threads = GmailApp.search(
      'to:' + email + ' OR from:' + email + ' after:' + after, 0, 10);
    out[email] = threads.map(function(t) {
      var msgs  = t.getMessages();
      var last  = msgs[msgs.length - 1];
      var isReply = msgs.some(function(m) {
        return m.getFrom().toLowerCase()
               .indexOf(email.toLowerCase()) !== -1;
      });
      return {
        subject:      t.getFirstMessageSubject(),
        messageCount: msgs.length,
        lastDate:     last.getDate().toISOString(),
        firstDate:    msgs[0].getDate().toISOString(),
        lastFrom:     last.getFrom(),
        snippet:      last.getPlainBody().substring(0, 250),
        isReply:      isReply
      };
    });
  });
  return ContentService
    .createTextOutput(JSON.stringify({ success: true, data: out }))
    .setMimeType(ContentService.MimeType.JSON);
}
